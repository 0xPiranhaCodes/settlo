// #208 capture app (local only): fakes, with THEME, ROUTE and UNREAD read
// from Documents/capture.txt (`key=value` lines).
import 'dart:io';

import 'package:flutter/material.dart';
import 'package:flutter_riverpod/flutter_riverpod.dart';
import 'package:path_provider/path_provider.dart';
import 'package:settlo/router/router.dart';
import 'package:settlo_core/settlo_core.dart';

import '../test/support/fake_expenses.dart';
import '../test/support/fake_home.dart';
import '../test/support/fake_session.dart';
import '../test/support/fake_settle.dart';
import '../test/support/fake_spaces.dart';

Future<void> main() async {
  WidgetsFlutterBinding.ensureInitialized();
  final file = File(
    '${(await getApplicationDocumentsDirectory()).path}/capture.txt',
  );
  final env = {
    for (final line in file.existsSync() ? file.readAsLinesSync() : <String>[])
      if (line.contains('='))
        line.substring(0, line.indexOf('=')): line.substring(
          line.indexOf('=') + 1,
        ),
  };
  final scope = fakeSessionApp(
    fakeSession,
    themeMode: env['THEME'] == 'light' ? ThemeMode.light : ThemeMode.dark,
    home: FakeHomeRepository(homeMain),
    overrides: [
      expensesRepositoryProvider.overrideWithValue(FakeExpensesRepository()),
      activityUnreadProvider.overrideWith(
        (ref) async => int.parse(env['UNREAD'] ?? '0'),
      ),
    ],
  ) as ProviderScope;
  final container = ProviderContainer(
    overrides: [
      ...scope.overrides.where(
        (override) =>
            !(env['ROUTE'] ?? '').startsWith('/settle') ||
            override.origin != spacesRepositoryProvider,
      ),
      if ((env['ROUTE'] ?? '').startsWith('/settle'))
        spacesRepositoryProvider.overrideWithValue(
          FakeSpacesRepository(settleSpaces()),
        ),
    ],
  );
  runApp(UncontrolledProviderScope(container: container, child: scope.child));
  final route = env['ROUTE'];
  if (route != null) {
    Future.delayed(const Duration(milliseconds: 800), () {
      container.read(routerProvider).go(route);
    });
  }
}
